-- 型指定
CREATE TABLE t_task (
    task_id BIGINT IDENTITY(1,1)    NOT NULL,
    task_name NVARCHAR(255)         NOT NULL,
    task_state_id INT               NOT NULL,
    update_at DATETIME              NOT NULL,
    user_id BIGINT                  NOT NULL,
    CONSTRAINT PK_t_task PRIMARY KEY (task_id),
);
