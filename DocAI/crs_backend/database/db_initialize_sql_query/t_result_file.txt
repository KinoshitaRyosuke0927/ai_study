-- 型指定
CREATE TABLE t_result_file (
    result_file_id BIGINT IDENTITY(1,1) NOT NULL,
    result_file_name NVARCHAR(255)      NOT NULL,
    original_file_id BIGINT             NOT NULL,
    task_id BIGINT                      NOT NULL,
    user_id BIGINT                      NOT NULL,
    CONSTRAINT PK_t_result_file PRIMARY KEY (result_file_id),
);
